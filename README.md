# ACCRUE_for_skewed

Repo for extending the ACCRUE cost function to skewed distributions.

# Dependencies:
- Anaconda
$ conda create -n env_ACCRUE python=3.11 -y
$ conda activate env_ACCRUE

# install core dependencies
$ conda install numpy=1.26 scipy scikit-learn -y
$ conda install torch
$ pip install twopiece

# install remaining packages
$ conda install skorch pytest
